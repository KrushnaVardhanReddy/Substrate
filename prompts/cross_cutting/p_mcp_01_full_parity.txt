    var args struct { Org string `json:"org"` }

    ctx := context.Background()
    result, err := store.SomeMethod(ctx, args.Org)

    b, _ := json.Marshal(result)
    return string(b), nil
  }

3. api/internal/mcp/server.go — ADD ServeSSE() METHOD
──────────────────────────────────────────────────────────────

Add a new method alongside ServeStdio():

  func (s *Server) ServeSSE(w http.ResponseWriter, r *http.Request) {
    // Set SSE headers
    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("Connection", "keep-alive")
    w.Header().Set("Access-Control-Allow-Origin", "*")

    flusher, ok := w.(http.Flusher)
    if !ok {
      http.Error(w, "SSE not supported", http.StatusInternalServerError)
      return
    }

    // Send endpoint discovery event (MCP SSE protocol)
    fmt.Fprintf(w, "event: endpoint\ndata: /mcp/message\n\n")
    flusher.Flush()

    // Keep connection alive with heartbeat
    ticker := time.NewTicker(5 * time.Second)
    defer ticker.Stop()
    for {
      select {
      case <-r.Context().Done():
        return
      case <-ticker.C:
        fmt.Fprintf(w, ": heartbeat\n\n")
        flusher.Flush()
      }
    }
  }

Add a new method ServeHTTPMessage() to handle POST /mcp/message:
  - Read JSON body → call s.Handle(body) → write response JSON.

  func mcpCall(t *testing.T, method string, id int, params any) map[string]any {
    payload := map[string]any{"jsonrpc":"2.0","id":id,"method":method}
    if params != nil { payload["params"] = params }
    body, _ := json.Marshal(payload)
    req, _ := http.NewRequest("POST", pMCPApiURL+"/mcp/message", bytes.NewReader(body))
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("Authorization", "Bearer "+pMCPToken)
    resp, err := http.DefaultClient.Do(req)
    require.NoError(t, err)
    defer resp.Body.Close()
    var result map[string]any
    json.NewDecoder(resp.Body).Decode(&result)
    return result
  }
